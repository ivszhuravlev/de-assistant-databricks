ensure_schema(spark, paths) -> None

load_paths(backend, output_space=config["output_space"]) -> RuntimePaths
    backend is "adls" or "dbfs".
    output_space is "eval" or "generated" from config_json. The generator is given "generated".
    paths.raw(*parts) uses the parts the spec declares.
    paths.table(layer, table) resolves Hive for this output_space.
    spark.table(paths.table(layer, table))
    Do not hard-code schema names.
    Do not use spark.read.format("delta").table.

append_metadata_rows(spark, paths, table_name, rows) -> str
    table_name is DATA_QUALITY or LAYER_RUN or PIPELINE_RUN.
    Rows must match that table. DATA_QUALITY gets data_quality_row dicts only.
    LAYER_RUN gets layer_run_row dicts only. PIPELINE_RUN gets pipeline_run_row dicts only.
    Mixing them fails. Observer Delta is shared; helpers pick the location.
    Correct: append_metadata_rows(spark, paths, DATA_QUALITY, [row, ...])
    Gold owns the single successful pipeline-run record. After all gold writes and checks pass,
    call exactly once with the shared pipeline_run_id from config_json:
        append_metadata_rows(spark, paths, PIPELINE_RUN, [
            pipeline_run_row(
                pipeline_run_id=pipeline_run_id,
                pipeline_name=config.get("project_name", ""),
                status="SUCCESS",
            )
        ])
    Bronze and silver do not append the SUCCESS pipeline_run row.
